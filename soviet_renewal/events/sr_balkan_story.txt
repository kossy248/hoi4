# The request is a choice about patronage, not an automatic alliance or annexation.
country_event = {
    picture = GFX_sr_campaign_balkan
    id = sr_balkan_story.1
    title = sr_bs_1_t
    desc = sr_bs_1_d
    is_triggered_only = yes
    fire_only_once = yes
    trigger = {
        tag = SOV
        has_country_flag = sr_bs_contacts_started
        exists = YUG
        YUG = { sr_balkan_safe_influence_target = yes }
    }
    option = {
        name = sr_bs_1_a
        trigger = { has_political_power > 59 }
        ai_chance = { base = 30 }
        custom_effect_tooltip = sr_bs_home_priority_tt
        add_political_power = -60
        set_country_flag = sr_bs_home_priority
        add_to_variable = { sr_story_trust = -4 }
        clamp_variable = { var = sr_story_trust min = 0 max = 100 }
        add_power_balance_value = { id = sr_union_balance value = -0.02 }
        add_timed_idea = { idea = sr_bs_homeland_grants days = 180 }
        if = {
            limit = { exists = YUG YUG = { sr_balkan_safe_influence_target = yes } }
            YUG = { add_popularity = { ideology = communism popularity = 0.12 } }
        }
        country_event = { id = sr_balkan_story.2 days = 45 }
    }
    option = {
        name = sr_bs_1_b
        trigger = { has_political_power > 19 }
        ai_chance = { base = 60 }
        add_political_power = -20
        set_country_flag = sr_bs_equal_partners

        clamp_variable = { var = sr_story_trust min = 0 max = 100 }
        add_power_balance_value = { id = sr_union_balance value = 0.02 }
        add_timed_idea = { idea = sr_bs_equal_exchange days = 180 }

            limit = { exists = YUG YUG = { sr_balkan_safe_influence_target = yes } }

    }
}
