# Territory is all currently defined core states of the specified original tags.
# No country scope is entered, so an annexed/nonexistent tag still works.
# Require some core territory to prevent empty map definitions from passing.
sr_formable_controls_poland = {
    any_state = { is_core_of = POL }
    NOT = {
        any_state = {
            is_core_of = POL
            OR = {
                NOT = { is_owned_by = ROOT }
                NOT = { is_controlled_by = ROOT }
            }
        }
    }
}
