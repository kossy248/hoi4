# Called once from new-game startup, in SOV country scope.
sr_initialize_union_effect = {
    if = {
        limit = { NOT = { has_country_flag = sr_union_initialized } }
        set_power_balance = {
            id = sr_union_balance
            left_side = sr_union_collapse_side
            right_side = sr_union_preservation_side
            set_default = yes
        }
        set_variable = { sr_resistance_russia = 0 }
        set_variable = { sr_resistance_ukraine = 0 }
        set_variable = { sr_resistance_belarus = 0 }
        set_variable = { sr_resistance_georgia = 0 }
        set_variable = { sr_resistance_armenia = 0 }
        set_variable = { sr_resistance_azerbaijan = 0 }
        set_variable = { sr_resistance_kazakhstan = 0 }
        set_variable = { sr_resistance_uzbekistan = 0 }
        set_variable = { sr_resistance_turkmenistan = 0 }
        set_variable = { sr_resistance_kyrgyzstan = 0 }
        set_variable = { sr_resistance_tajikistan = 0 }
        set_variable = { sr_resistance_estonia = 0 }
        set_variable = { sr_resistance_latvia = 0 }
        set_variable = { sr_resistance_lithuania = 0 }
        set_variable = { sr_resistance_moldavia = 0 }
        set_country_flag = sr_union_initialized
    }
}

# Custom domestic unrest, separate from occupation resistance. Soviet cores stay intact.

sr_union_resistance_monthly_effect = {

        limit = { sr_union_russia_represented = yes }
        if = {
            limit = { power_balance_value = { id = sr_union_balance value < -0.60 } }
            add_to_variable = { sr_resistance_russia = 10 }
        }
        else_if = {
            limit = { power_balance_value = { id = sr_union_balance value > 0.20 } }
            subtract_from_variable = { sr_resistance_russia = 5 }
        }
        else = { subtract_from_variable = { sr_resistance_russia = 2 } }
        clamp_variable = { var = sr_resistance_russia min = 0 max = 100 }
        if = {
            limit = { check_variable = { sr_resistance_russia = 80 } }
            random_owned_state = {
                limit = {
                    is_controlled_by = ROOT
                    is_core_of = SOV
                    NOT = {
                        OR = {
                            is_core_of = UKR
                            is_core_of = BLR
                            is_core_of = GEO
                            is_core_of = ARM
                            is_core_of = AZR
                            is_core_of = KAZ
                            is_core_of = UZB
                            is_core_of = TMS
                            is_core_of = KYR

                            is_core_of = EST
                            is_core_of = LAT

# Re-evaluate immediately after recovery decisions as well as monthly updates.
sr_update_union_resistance_ideas_effect = {
    if = {
        limit = {
            sr_union_has_strikes = yes
            NOT = { has_idea = sr_union_mass_strikes }
        }
        add_ideas = sr_union_mass_strikes
    }
    else_if = {
        limit = {
            NOT = { sr_union_has_strikes = yes }
            has_idea = sr_union_mass_strikes
        }
        remove_ideas = sr_union_mass_strikes
    }
    if = {
        limit = {
            sr_union_has_sabotage = yes
            NOT = { has_idea = sr_union_sabotage_networks }
        }
        add_ideas = sr_union_sabotage_networks
    }
    else_if = {
        limit = {
            NOT = { sr_union_has_sabotage = yes }
            has_idea = sr_union_sabotage_networks
        }
        remove_ideas = sr_union_sabotage_networks
    }
}
